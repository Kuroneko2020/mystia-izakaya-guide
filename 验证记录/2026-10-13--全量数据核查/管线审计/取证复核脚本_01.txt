def read(rel):
    return (root / rel).read_text(encoding='utf-8-sig')

def load(rel):
    return json.loads(read(rel))

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(stem, data):
    # Resume after a collector failure without duplicating already saved identical evidence.
    if os.environ.get('EVIDENCE_REUSE_MATCHING') == '1':
        for existing_path in sorted(dest.glob(stem + '_*.json')):
            if json.loads(existing_path.read_text(encoding='utf-8')) == data:
                print('Preserved existing evidence:', existing_path)
                return existing_path
    for i in range(1, 10000):
        path = dest / f'{stem}_{i:02}.json'
        try:
            with path.open('x', encoding='utf-8', newline='\n') as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
                f.write('\n')
            print(path)
            return path
        except FileExistsError:
            continue
    raise RuntimeError('No available create-new evidence filename')

def pure_functions(rel, names, context):
    tree = ast.parse(read(rel), filename=rel)
    for node in tree.body:
        if isinstance(node, ast.FunctionDef) and node.name in names:
            exec(compile(ast.Module(body=[node], type_ignores=[]), rel, 'exec'), context)
    return tree
